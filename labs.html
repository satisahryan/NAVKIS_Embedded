sketch.ino
// Lab 1 - Hello, GPIO!  (ESP32 DevKitC V4 in Wokwi)
// An LED on GPIO 26 (through a 220 ohm resistor) blinks once a second.

const int LED_PIN = 26;        // the GPIO pin our LED is wired to

void setup() {
  Serial.begin(115200);        // open the serial monitor (115200 baud)
  pinMode(LED_PIN, OUTPUT);    // configure GPIO 26 as an OUTPUT
  Serial.println("Lab 1: Hello, GPIO!");
}

void loop() {
  digitalWrite(LED_PIN, HIGH); // 3.3 V on the pin -> LED ON
  Serial.println("LED ON");
  delay(500);                  // wait 500 ms (the CPU does nothing else!)

  digitalWrite(LED_PIN, LOW);  // 0 V on the pin -> LED OFF
  Serial.println("LED OFF");
  delay(500);
}

Solution to 3 — the millis() pattern
solution_millis.ino
// Lab 1 - Try-this solution: two LEDs at two different rates WITHOUT delay()
// Red LED on GPIO 26 toggles every 500 ms, green LED on GPIO 27 every 130 ms.

const int RED_PIN = 26;
const int GREEN_PIN = 27;

unsigned long lastRed = 0, lastGreen = 0;   // time of the last toggle
bool redOn = false, greenOn = false;

void setup() {
  pinMode(RED_PIN, OUTPUT);
  pinMode(GREEN_PIN, OUTPUT);
}

void loop() {
  unsigned long now = millis();             // ms since the board started

  if (now - lastRed >= 500) {               // is it time to toggle red?
    lastRed = now;
    redOn = !redOn;
    digitalWrite(RED_PIN, redOn);
  }
  if (now - lastGreen >= 130) {             // is it time to toggle green?
    lastGreen = now;
    greenOn = !greenOn;
    digitalWrite(GREEN_PIN, greenOn);
  }
  // loop() never blocks, so the CPU is free to do other work here
}

Big idea: instead of waiting, remember when you last acted and check the clock. The loop stays free — this is cooperative multitasking.

sketch.ino (interrupt version)
// Lab 2 - Buttons & Interrupts
// Push-button on GPIO 4 (other leg to GND), LED on GPIO 26.
// Every press toggles the LED and increments a counter - handled by an INTERRUPT.

const int BUTTON_PIN = 4;
const int LED_PIN = 26;

volatile bool pressed = false;        // set inside the ISR, read in loop()
volatile unsigned long lastIsrMs = 0; // for debouncing

// Interrupt Service Routine: keep it SHORT. No Serial.print, no delay here.
void IRAM_ATTR onButton() {
  unsigned long now = millis();
  if (now - lastIsrMs > 200) {        // ignore contact bounce (< 200 ms)
    pressed = true;
    lastIsrMs = now;
  }
}

int count = 0;
bool ledOn = false;

void setup() {
  Serial.begin(115200);
  pinMode(LED_PIN, OUTPUT);
  pinMode(BUTTON_PIN, INPUT_PULLUP);  // internal pull-up: idle = HIGH, pressed = LOW
  attachInterrupt(digitalPinToInterrupt(BUTTON_PIN), onButton, FALLING);
  Serial.println("Lab 2: press the button!");
}

void loop() {
  if (pressed) {                      // the ISR raised the flag
    pressed = false;
    count++;
    ledOn = !ledOn;
    digitalWrite(LED_PIN, ledOn);
    Serial.printf("Press #%d -> LED %s\n", count, ledOn ? "ON" : "OFF");
  }
  // The CPU is free here - e.g. blink something else, read sensors...
}
Polling version, for comparison
polling.ino
// Lab 2 - Polling version (for comparison). Same wiring as the interrupt version.
const int BUTTON_PIN = 4;
const int LED_PIN = 26;

void setup() {
  Serial.begin(115200);
  pinMode(LED_PIN, OUTPUT);
  pinMode(BUTTON_PIN, INPUT_PULLUP);
}

void loop() {
  int state = digitalRead(BUTTON_PIN);     // ask the pin, again and again...
  digitalWrite(LED_PIN, state == LOW);     // LED on while the button is held
  delay(10);
}

Try this

  1. Change the debounce window from 200 ms to 0. Click quickly — do you ever see a press counted twice? That is contact bounce.
  2. Why must pressed be volatile? (Hint: who else changes it?)
  3. Add delay(3000) at the end of loop() in both versions. Press the button during the delay. Which version still notices the press?
  4. Count only long presses (held > 1 s). Deep dive
Answers
  1. Real switches “chatter” for a few milliseconds, creating several edges. The time window ignores them.
  2. It is changed inside the ISR, outside the normal flow of loop(). volatile stops the compiler from caching it in a register.
  3. The interrupt version: the ISR sets the flag even while loop() sleeps, and the press is handled once the delay ends. The polling version misses it completely.
  4. Attach the interrupt on CHANGE, store the press time on LOW and compare on HIGH; or poll and time with millis().

// Lab 3 - The Analog World: ADC in, PWM out
// Potentiometer wiper -> GPIO 34 (ADC1), LED on GPIO 26 dimmed with PWM.
// Open the Serial Plotter style output: each line prints "raw volts duty".

const int POT_PIN = 34;   // ADC1 channel (input-only pin)
const int LED_PIN = 26;

void setup() {
  Serial.begin(115200);
  pinMode(LED_PIN, OUTPUT);
  analogReadResolution(12);           // 12-bit ADC: 0 ... 4095
}

void loop() {
  int raw = analogRead(POT_PIN);              // 0 ... 4095
  float volts = raw * 3.3 / 4095.0;           // convert to voltage
  int duty = map(raw, 0, 4095, 0, 255);       // scale to 8-bit PWM duty
  analogWrite(LED_PIN, duty);                 // PWM: brightness follows the knob

  Serial.printf("raw:%d volts:%.2f duty:%d\n", raw, volts, duty);
  delay(100);
}

solution_nightlamp.ino
// Lab 3 - Try-this solution: automatic night lamp
// LDR module AO -> GPIO 35. In Wokwi the AO value RISES when it gets darker.
const int LDR_PIN = 35;
const int LED_PIN = 26;
int DARK_THRESHOLD = 2000;   // calibrate: watch the printed values, then choose

void setup() {
  Serial.begin(115200);
  pinMode(LED_PIN, OUTPUT);
}

void loop() {
  int light = analogRead(LDR_PIN);
  bool dark = light > DARK_THRESHOLD;
  digitalWrite(LED_PIN, dark);
  Serial.printf("ldr:%d lamp:%s\n", light, dark ? "ON" : "OFF");
  delay(200);
}

sketch.ino
// Lab 4 - Sensor + Display over I2C
// DHT22 data -> GPIO 15, SSD1306 OLED on I2C (SDA = GPIO 21, SCL = GPIO 22),
// alarm LED on GPIO 26 switches on above TEMP_ALARM.
#include <Wire.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>
#include "DHTesp.h"

const int DHT_PIN = 15;
const int LED_PIN = 26;
const float TEMP_ALARM = 35.0;      // degrees C

Adafruit_SSD1306 oled(128, 64, &Wire, -1);
DHTesp dht;

void setup() {
  Serial.begin(115200);
  pinMode(LED_PIN, OUTPUT);
  dht.setup(DHT_PIN, DHTesp::DHT22);

  if (!oled.begin(SSD1306_SWITCHCAPVCC, 0x3C)) {   // 0x3C = OLED I2C address
    Serial.println("OLED not found - check SDA/SCL wiring");
    while (true) delay(10);
  }
  oled.clearDisplay();
  oled.setTextColor(SSD1306_WHITE);
}

void loop() {
  TempAndHumidity d = dht.getTempAndHumidity();
  if (dht.getStatus() != 0) {
    Serial.println("DHT error: " + String(dht.getStatusString()));
    delay(2000);
    return;
  }
  bool alarm = d.temperature > TEMP_ALARM;
  digitalWrite(LED_PIN, alarm);

  Serial.printf("temp:%.1f hum:%.1f alarm:%d\n", d.temperature, d.humidity, alarm);

  oled.clearDisplay();
  oled.setTextSize(1);
  oled.setCursor(0, 0);
  oled.println("NAVKIS Weather Node");
  oled.setTextSize(2);
  oled.setCursor(0, 18);
  oled.printf("%.1f C\n", d.temperature);
  oled.printf("%.0f %%RH", d.humidity);
  if (alarm) {
    oled.setTextSize(1);
    oled.setCursor(0, 56);
    oled.print("!! HIGH TEMPERATURE !!");
  }
  oled.display();
  delay(2000);                        // DHT22 can be read at most every 2 s
}
The DHT22’s pin is labelled SDA in Wokwi, but it is not I²C — it uses its own single-wire timing protocol. Only the OLED is on the I²C bus here.

// Lab 5 - Your ESP32 on the Internet (MQTT)
// DHT22 -> GPIO 15, LED -> GPIO 26. Joins Wokwi's virtual Wi-Fi, publishes
// readings every 5 s, and listens for ON/OFF commands from the class dashboard.
#include <WiFi.h>
#include <PubSubClient.h>
#include "DHTesp.h"

// ======= CHANGE THIS: your name or USN (letters/numbers only, no spaces) =======
const char* STUDENT = "your-name";
// ===============================================================================

const char* MQTT_BROKER = "broker.hivemq.com";   // free public broker
const int   MQTT_PORT   = 1883;
const char* TOPIC_ROOT  = "navkis/ews2026";      // must match the dashboard

const int DHT_PIN = 15;
const int LED_PIN = 26;

WiFiClient net;
PubSubClient mqtt(net);
DHTesp dht;
String dataTopic, ledTopic;
unsigned long lastSend = 0;

// Called whenever a message arrives on a topic we subscribed to
void onMessage(char* topic, byte* payload, unsigned int length) {
  String msg;
  for (unsigned int i = 0; i < length; i++) msg += (char)payload[i];
  Serial.printf("Message on %s: %s\n", topic, msg.c_str());
  if (msg == "ON")  digitalWrite(LED_PIN, HIGH);
  if (msg == "OFF") digitalWrite(LED_PIN, LOW);
}

void connectWiFi() {
  Serial.print("Connecting to Wi-Fi");
  WiFi.begin("Wokwi-GUEST", "", 6);       // open network, channel 6
  while (WiFi.status() != WL_CONNECTED) {
    delay(100);
    Serial.print(".");
  }
  Serial.print(" connected! IP = ");
  Serial.println(WiFi.localIP());
}

void connectMQTT() {
  while (!mqtt.connected()) {
    String clientId = "navkis-" + String(STUDENT) + "-" + String(random(0xffff), HEX);
    Serial.print("Connecting to MQTT broker... ");
    if (mqtt.connect(clientId.c_str())) {
      Serial.println("connected!");
      mqtt.subscribe(ledTopic.c_str());
      Serial.println("Listening on " + ledTopic);
    } else {
      Serial.printf("failed (state %d), retrying in 2 s\n", mqtt.state());
      delay(2000);
    }
  }
}

void setup() {
  Serial.begin(115200);
  pinMode(LED_PIN, OUTPUT);
  dht.setup(DHT_PIN, DHTesp::DHT22);
  randomSeed(micros());

  dataTopic = String(TOPIC_ROOT) + "/" + STUDENT + "/data";
  ledTopic  = String(TOPIC_ROOT) + "/" + STUDENT + "/led";

  connectWiFi();
  mqtt.setServer(MQTT_BROKER, MQTT_PORT);
  mqtt.setCallback(onMessage);
}

void loop() {
  if (!mqtt.connected()) connectMQTT();
  mqtt.loop();                            // handle incoming messages

  if (millis() - lastSend > 5000) {       // publish every 5 seconds
    lastSend = millis();
    TempAndHumidity d = dht.getTempAndHumidity();
    char json[96];
    snprintf(json, sizeof(json), "{\"t\":%.1f,\"h\":%.1f,\"led\":%d}",
             d.temperature, d.humidity, digitalRead(LED_PIN));
    mqtt.publish(dataTopic.c_str(), json);
    Serial.printf("Published to %s: %s\n", dataTopic.c_str(), json);
  }
}

What is happening?

  1. Wokwi provides a virtual access point, Wokwi-GUEST (no password), connected to the real Internet.
  2. Your ESP32 opens a TCP connection to the public broker broker.hivemq.com on port 1883.
  3. Every 5 s it publishes a small JSON message to navkis/ews2026/<your-name>/data.
  4. It subscribes to navkis/ews2026/<your-name>/led. When anyone publishes ON or OFF there, your LED obeys.
  5. The Live Dashboard subscribes to navkis/ews2026/# and shows everyone at once.
A public broker is like a public notice board: anyone can read your topic. Never send passwords or personal data this way. Real products use TLS and a username and password.

// Lab 6 (Deep dive) - Real-time multitasking with FreeRTOS
// Three independent tasks share one CPU, like processes in your OS course:
//   blinkTask  - toggles the LED on GPIO 26 every 250 ms
//   sensorTask - reads the potentiometer (GPIO 34) and sends values to a QUEUE
//   printTask  - waits on the queue and prints whatever arrives
// Potentiometer wiper -> GPIO 34, LED -> GPIO 26 (same wiring as Lab 3).

const int LED_PIN = 26;
const int POT_PIN = 34;

QueueHandle_t potQueue;               // FIFO between sensorTask and printTask

void blinkTask(void* param) {
  pinMode(LED_PIN, OUTPUT);
  while (true) {
    digitalWrite(LED_PIN, !digitalRead(LED_PIN));
    vTaskDelay(pdMS_TO_TICKS(250));   // sleep this task; others keep running
  }
}

void sensorTask(void* param) {
  while (true) {
    int value = analogRead(POT_PIN);
    xQueueSend(potQueue, &value, 0);  // drop the sample if the queue is full
    vTaskDelay(pdMS_TO_TICKS(500));
  }
}

void printTask(void* param) {
  int value;
  while (true) {
    if (xQueueReceive(potQueue, &value, portMAX_DELAY)) {  // block until data
      Serial.printf("[core %d] pot = %d\n", xPortGetCoreID(), value);
    }
  }
}

void setup() {
  Serial.begin(115200);
  potQueue = xQueueCreate(10, sizeof(int));   // holds up to 10 ints

  //                       function    name      stack  param prio handle
  xTaskCreate(blinkTask,  "blink",   2048,  NULL, 1,   NULL);
  xTaskCreate(sensorTask, "sensor",  2048,  NULL, 2,   NULL);
  xTaskCreate(printTask,  "print",   4096,  NULL, 1,   NULL);
}

void loop() {
  vTaskDelay(pdMS_TO_TICKS(1000));    // loop() is itself just another task
}

// Mini-challenge - Smart Street Light (reference solution)
const int LDR_PIN = 35;
const int PIR_PIN = 27;
const int LAMP_PIN = 26;
const int DARK_THRESHOLD = 2000;      // calibrate from the printed ldr values
const unsigned long HOLD_MS = 5000;

unsigned long lastMotion = 0;
String lastState = "";

void setup() {
  Serial.begin(115200);
  pinMode(PIR_PIN, INPUT);
  pinMode(LAMP_PIN, OUTPUT);
}

void loop() {
  bool dark = analogRead(LDR_PIN) > DARK_THRESHOLD;
  if (digitalRead(PIR_PIN) == HIGH) lastMotion = millis();
  bool recentMotion = (lastMotion != 0) && (millis() - lastMotion < HOLD_MS);

  String state;
  if (!dark)             { analogWrite(LAMP_PIN, 0);   state = "DAY"; }
  else if (recentMotion) { analogWrite(LAMP_PIN, 255); state = "NIGHT-FULL"; }
  else                   { analogWrite(LAMP_PIN, 50);  state = "NIGHT-DIM"; }

  if (state != lastState) {           // print only on change
    Serial.println(state);
    lastState = state;
  }
  delay(50);
}
